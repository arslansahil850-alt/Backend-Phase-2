Overview: This single executable backend server combines all concepts learned across Days 1 through 14 into one cohesive production structure.

Included Modules & Technical Highlights:

  • Environment Configuration: Loads variables via dotenv with missing variable validation.
  • Global Middleware: Request logger, JSON parser, and request timestamps.
  • RESTful Endpoints: Plural endpoints (/api/v1/tasks) with full CRUD (GET, POST, PUT, PATCH, DELETE).
  • Advanced Filtering & Pagination: Slice-based pagination with page, limit, and status filters.
  • Centralized Error Handling: Custom AppError class and 4-parameter error middleware.
capstone-server.js 
/**
 * PHASE 2 CAPSTONE: COMPLETE PRODUCTION-GRADE EXPRESS REST API
 * Combines: Node.js, Express, Middleware, Routing, CRUD, REST Principles, 
 * Error Handling, Pagination, and Dotenv Security.
 */

require('dotenv').config();
const express = require('express');
const app = express();

// 1. ENVIRONMENT CONFIGURATION & VALIDATION
const PORT = process.env.PORT || 3000;
const API_KEY = process.env.API_KEY || 'arslan-secret-key';

// 2. CUSTOM ERROR CLASS (Day 12 & 13)
class AppError extends Error {
    constructor(message, statusCode) {
        super(message);
        this.statusCode = statusCode;
        this.status = `${statusCode}`.startsWith('4') ? 'fail' : 'error';
        this.isOperational = true;
        Error.captureStackTrace(this, this.constructor);
    }
}

// 3. GLOBAL MIDDLEWARE PIPELINE (Day 09 & 11)
app.use(express.json());

// Custom Logger Middleware
app.use((req, res, next) => {
    req.requestTime = new Date().toISOString();
    console.log(`[${req.requestTime}] ${req.method} -> ${req.originalUrl}`);
    next();
});

// API Key Security Guard Middleware (Day 14)
const authenticateKey = (req, res, next) => {
    const key = req.headers['x-api-key'];
    if (!key || key !== API_KEY) {
        return next(new AppError('Unauthorized: Missing or invalid X-API-KEY header', 401));
    }
    next();
};

// 4. IN-MEMORY DATABASE STORE (Day 10)
let tasks = [
    { id: 1, title: 'Learn Node.js Basics', status: 'completed', priority: 'high', createdAt: '2026-01-01' },
    { id: 2, title: 'Master Express Routing', status: 'completed', priority: 'medium', createdAt: '2026-01-05' },
    { id: 3, title: 'Build REST APIs', status: 'in-progress', priority: 'high', createdAt: '2026-01-10' },
    { id: 4, title: 'Implement Dotenv & Error Handlers', status: 'pending', priority: 'medium', createdAt: '2026-01-14' },
    { id: 5, title: 'Deploy Phase 2 Project', status: 'pending', priority: 'low', createdAt: '2026-01-15' }
];

// 5. RESTFUL ROUTES (Day 08, 10, 13)

// GET /api/v1/tasks (With Pagination & Filtering)
app.get('/api/v1/tasks', (req, res) => {
    let results = [...tasks];

    // Filtering by status
    if (req.query.status) {
        results = results.filter(t => t.status.toLowerCase() === req.query.status.toLowerCase());
    }

    // Pagination
    const page = parseInt(req.query.page) || 1;
    const limit = parseInt(req.query.limit) || 2;
    const startIndex = (page - 1) * limit;
    const endIndex = page * limit;

    const paginatedTasks = results.slice(startIndex, endIndex);

    res.status(200).json({
        success: true,
        page,
        limit,
        totalItems: results.length,
        totalPages: Math.ceil(results.length / limit),
        data: paginatedTasks
    });
});

// GET /api/v1/tasks/:id
app.get('/api/v1/tasks/:id', (req, res, next) => {
    const task = tasks.find(t => t.id === parseInt(req.params.id));
    if (!task) {
        return next(new AppError('Task not found with that ID', 404));
    }
    res.status(200).json({ success: true, data: task });
});

// POST /api/v1/tasks (Protected)
app.post('/api/v1/tasks', authenticateKey, (req, res, next) => {
    const { title, priority } = req.body;
    if (!title) {
        return next(new AppError('Please provide a task title', 400));
    }

    const newTask = {
        id: tasks.length + 1,
        title,
        status: 'pending',
        priority: priority || 'medium',
        createdAt: new Date().toISOString().split('T')[0]
    };

    tasks.push(newTask);
    res.status(201).json({ success: true, message: 'Task created successfully', data: newTask });
});

// PUT /api/v1/tasks/:id (Replace Complete Task)
app.put('/api/v1/tasks/:id', authenticateKey, (req, res, next) => {
    const id = parseInt(req.params.id);
    const index = tasks.findIndex(t => t.id === id);
    if (index === -1) {
        return next(new AppError('Task not found for replacement', 404));
    }

    const { title, status, priority } = req.body;
    if (!title || !status) {
        return next(new AppError('Title and status are required for full PUT replacement', 400));
    }

    tasks[index] = { id, title, status, priority: priority || 'medium', createdAt: tasks[index].createdAt };
    res.status(200).json({ success: true, message: 'Task replaced successfully', data: tasks[index] });
});

// PATCH /api/v1/tasks/:id (Partial Update)
app.patch('/api/v1/tasks/:id', authenticateKey, (req, res, next) => {
    const id = parseInt(req.params.id);
    const task = tasks.find(t => t.id === id);
    if (!task) {
        return next(new AppError('Task not found for partial update', 404));
    }

    if (req.body.title) task.title = req.body.title;
    if (req.body.status) task.status = req.body.status;
    if (req.body.priority) task.priority = req.body.priority;

    res.status(200).json({ success: true, message: 'Task updated successfully', data: task });
});

// DELETE /api/v1/tasks/:id
app.delete('/api/v1/tasks/:id', authenticateKey, (req, res, next) => {
    const index = tasks.findIndex(t => t.id === parseInt(req.params.id));
    if (index === -1) {
        return next(new AppError('Task not found for deletion', 404));
    }

    tasks.splice(index, 1);
    res.status(200).json({ success: true, message: 'Task deleted successfully' });
});

// 6. UNHANDLED ROUTE CATCH-ALL (404)
app.all('*', (req, res, next) => {
    next(new AppError(`Cannot find ${req.originalUrl} on this server!`, 404));
});

// 7. CENTRALIZED ERROR HANDLING MIDDLEWARE (Day 12)
app.use((err, req, res, next) => {
    err.statusCode = err.statusCode || 500;
    err.status = err.status || 'error';

    res.status(err.statusCode).json({
        success: false,
        status: err.status,
        message: err.message,
        stack: process.env.NODE_ENV === 'development' ? err.stack : undefined
    });
});

// 8. START SERVER
app.listen(PORT, () => {
    console.log(`🚀 Capstone REST API server running at http://localhost:${PORT}`);
});
